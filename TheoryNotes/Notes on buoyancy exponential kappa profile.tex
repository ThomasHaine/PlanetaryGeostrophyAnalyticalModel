\section{Problem statement}

Consider the one-dimensional vertical boundary-value problem
\begin{equation}
E k\,\partial_z\!\left(\kappa(z)\,\partial_z b\right)
=
\Gamma \left(b-B\right),
\label{eq:main}
\end{equation}
on the interval
\[
-H \le z \le 0,
\]
with boundary conditions
\begin{align}
b(0) &= b_s, \label{eq:bc_surface}\\
\partial_z b(-H) &= 0. \label{eq:bc_bottom}
\end{align}

Here \(B=B(x,y,z)\) is a prescribed buoyancy profile, \(b_s=b_s(x,y)\) is the imposed surface buoyancy, \(\Gamma\) is a relaxation coefficient, and \(E k\) is a constant prefactor.

We take the vertical diffusivity to be
\begin{equation}
\kappa(z)
=
a+\beta \exp\!\left[-\frac{z+H}{h}\right],
\label{eq:kappa}
\end{equation}
where \(a>0\), \(h>0\), and \(\beta\) is a constant. I use the symbol \(\beta\) for the exponential coefficient in \(\kappa\) to avoid notational conflict with the buoyancy solution \(b\).

The horizontal dependence \(x,y\) is parametric in this vertical problem, so it will be suppressed in most of the derivation.

Define
\begin{equation}
\mu = \frac{\Gamma}{E k}.
\label{eq:mu}
\end{equation}
Then \eqref{eq:main} may be written as
\begin{equation}
\partial_z\!\left(\kappa b_z\right)-\mu b
=
-\mu B.
\label{eq:sl_form}
\end{equation}

This is a forced Sturm--Liouville problem with mixed Dirichlet--Neumann boundary conditions.

\section{Homogeneous equation}

The associated homogeneous equation is
\begin{equation}
\partial_z\!\left(\kappa b_z\right)-\mu b =0.
\label{eq:homogeneous}
\end{equation}

Using
\[
\kappa(z)=a+\beta \exp\!\left[-\frac{z+H}{h}\right],
\]
introduce the transformed variable
\begin{equation}
X(z)
=
-\frac{\beta}{a}\exp\!\left[-\frac{z+H}{h}\right].
\label{eq:X_def}
\end{equation}
Then
\begin{equation}
\kappa(z)=a(1-X).
\end{equation}

Also define the dimensionless parameter
\begin{equation}
\nu
=
h\sqrt{\frac{\mu}{a}}
=
h\sqrt{\frac{\Gamma}{E k\,a}}.
\label{eq:nu_def}
\end{equation}

With these definitions, a convenient basis of homogeneous solutions is
\begin{align}
y_+(z)
&=
X(z)^{\nu}\,
{}_2F_1\!\left(\nu,\nu+1;2\nu+1;X(z)\right),
\label{eq:yplus}
\\[0.5em]
y_-(z)
&=
X(z)^{-\nu}\,
{}_2F_1\!\left(-\nu,1-\nu;1-2\nu;X(z)\right).
\label{eq:yminus}
\end{align}

Therefore the general homogeneous solution is
\begin{equation}
b_h(z)=C_+ y_+(z)+C_- y_-(z).
\label{eq:general_homogeneous}
\end{equation}

The function \({}_2F_1\) is the Gauss hypergeometric function. When \(\beta>0\), one has \(X<0\), and the real-valued branch should be chosen consistently.

\section{Construction of boundary-adapted homogeneous solutions}

We now construct two special homogeneous solutions.

Let \(\phi(z)\) solve
\begin{equation}
L\phi=0,
\qquad
\phi_z(-H)=0,
\label{eq:phi_def}
\end{equation}
where
\begin{equation}
L f := \partial_z(\kappa f_z)-\mu f.
\end{equation}

Similarly, let \(\psi(z)\) solve
\begin{equation}
L\psi=0,
\qquad
\psi(0)=0.
\label{eq:psi_def}
\end{equation}

Using the homogeneous basis \(y_\pm\), a convenient choice is
\begin{equation}
\phi(z)
=
y_-'(-H)y_+(z)-y_+'(-H)y_-(z),
\label{eq:phi_explicit}
\end{equation}
because
\[
\phi_z(-H)
=
y_-'(-H)y_+'(-H)-y_+'(-H)y_-'(-H)
=
0.
\]

Likewise, a convenient choice for \(\psi\) is
\begin{equation}
\psi(z)
=
y_-(0)y_+(z)-y_+(0)y_-(z),
\label{eq:psi_explicit}
\end{equation}
because
\[
\psi(0)
=
y_-(0)y_+(0)-y_+(0)y_-(0)
=
0.
\]

\section{Green's function}

The Green's function \(G(z,\zeta)\) is defined by
\begin{equation}
L_z G(z,\zeta)=\delta(z-\zeta),
\label{eq:green_def}
\end{equation}
with boundary conditions
\begin{align}
G(0,\zeta)&=0,\\
\partial_z G(-H,\zeta)&=0.
\end{align}

For a Sturm--Liouville operator of the form
\[
L f = \partial_z(\kappa f_z)-\mu f,
\]
the weighted Wronskian
\begin{equation}
W
=
\kappa(z)
\left[
\phi(z)\psi_z(z)-\phi_z(z)\psi(z)
\right]
\label{eq:wronskian}
\end{equation}
is independent of \(z\).

The Green's function is then
\begin{equation}
G(z,\zeta)
=
\begin{cases}
\dfrac{\phi(z)\psi(\zeta)}{W},
& z<\zeta,\\[1.2em]
\dfrac{\phi(\zeta)\psi(z)}{W},
& z>\zeta.
\end{cases}
\label{eq:green}
\end{equation}

This convention corresponds to the jump condition
\begin{equation}
\kappa(\zeta)
\left[
G_z(\zeta^+,\zeta)-G_z(\zeta^-,\zeta)
\right]
=1.
\end{equation}

\section{Solution of the forced problem}

The inhomogeneous problem is
\begin{equation}
L b = -\mu B,
\end{equation}
with
\[
b(0)=b_s,
\qquad
b_z(-H)=0.
\]

The boundary condition at the surface is nonhomogeneous, so we decompose the solution into a boundary-forced part and an internally forced part.

The solution satisfying the surface condition \(b(0)=b_s\) and the bottom Neumann condition is
\begin{equation}
b_{\mathrm{surf}}(z)
=
b_s\,\frac{\phi(z)}{\phi(0)}.
\label{eq:surface_solution}
\end{equation}

The solution forced by the interior source \(-\mu B\) and satisfying homogeneous boundary conditions is
\begin{equation}
b_{\mathrm{int}}(z)
=
-\mu
\int_{-H}^{0}
G(z,\zeta)B(\zeta)\,d\zeta.
\label{eq:interior_solution}
\end{equation}

Therefore the full solution is
\begin{equation}
\boxed{
b(z)
=
b_s\,\frac{\phi(z)}{\phi(0)}
-
\mu
\int_{-H}^{0}
G(z,\zeta)B(\zeta)\,d\zeta
}
\label{eq:solution_1d}
\end{equation}
or, restoring the horizontal dependence,
\begin{equation}
\boxed{
b(x,y,z)
=
b_s(x,y)\,\frac{\phi(z)}{\phi(0)}
-
\frac{\Gamma}{E k}
\int_{-H}^{0}
G(z,\zeta)B(x,y,\zeta)\,d\zeta.
}
\label{eq:solution_full}
\end{equation}

\section{Summary of definitions}

The diffusivity is
\begin{equation}
\kappa(z)
=
a+\beta \exp\!\left[-\frac{z+H}{h}\right].
\end{equation}

Define
\begin{equation}
\mu = \frac{\Gamma}{E k},
\qquad
\nu
=
h\sqrt{\frac{\mu}{a}}
=
h\sqrt{\frac{\Gamma}{E k\,a}},
\end{equation}
and
\begin{equation}
X(z)
=
-\frac{\beta}{a}\exp\!\left[-\frac{z+H}{h}\right].
\end{equation}

The homogeneous basis functions are
\begin{align}
y_+(z)
&=
X(z)^{\nu}\,
{}_2F_1\!\left(\nu,\nu+1;2\nu+1;X(z)\right),
\\[0.5em]
y_-(z)
&=
X(z)^{-\nu}\,
{}_2F_1\!\left(-\nu,1-\nu;1-2\nu;X(z)\right).
\end{align}

The bottom-adapted and surface-adapted homogeneous solutions are
\begin{align}
\phi(z)
&=
y_-'(-H)y_+(z)-y_+'(-H)y_-(z),
\\[0.5em]
\psi(z)
&=
y_-(0)y_+(z)-y_+(0)y_-(z).
\end{align}

The weighted Wronskian is
\begin{equation}
W
=
\kappa(z)
\left[
\phi(z)\psi_z(z)-\phi_z(z)\psi(z)
\right],
\end{equation}
which is independent of \(z\).

The Green's function is
\begin{equation}
G(z,\zeta)
=
\begin{cases}
\dfrac{\phi(z)\psi(\zeta)}{W},
& z<\zeta,\\[1.2em]
\dfrac{\phi(\zeta)\psi(z)}{W},
& z>\zeta.
\end{cases}
\end{equation}

Thus,
\begin{equation}
\boxed{
b(x,y,z)
=
b_s(x,y)\frac{\phi(z)}{\phi(0)}
-
\frac{\Gamma}{E k}
\int_{-H}^{0}
G(z,\zeta)B(x,y,\zeta)\,d\zeta.
}
\end{equation}

\section{Constant-diffusivity limit}

If \(\beta=0\), then \(\kappa=a\), and the governing equation becomes
\begin{equation}
a b_{zz}-\mu b=-\mu B.
\end{equation}

Define
\begin{equation}
\lambda = \sqrt{\frac{\mu}{a}}
=
\sqrt{\frac{\Gamma}{E k\,a}}.
\end{equation}

The homogeneous solutions are simply
\begin{equation}
b_h(z)=C_1\cosh\!\left[\lambda(z+H)\right]
+
C_2\sinh\!\left[\lambda(z+H)\right].
\end{equation}

The bottom condition \(b_z(-H)=0\) selects
\begin{equation}
\phi(z)=\cosh\!\left[\lambda(z+H)\right],
\end{equation}
while the surface-vanishing solution may be chosen as
\begin{equation}
\psi(z)=\sinh(\lambda z),
\end{equation}
up to an arbitrary multiplicative constant. The same Green's-function construction applies.

This limiting case is useful for checking signs and normalization.

\section{Remarks}

\begin{itemize}
    \item The operator
    \[
    L=\partial_z(\kappa \partial_z)-\mu
    \]
    is self-adjoint on \([-H,0]\) with the mixed Dirichlet--Neumann boundary conditions used above.

    \item The Green's function is symmetric:
    \[
    G(z,\zeta)=G(\zeta,z),
    \]
    provided the same normalization convention is used.

    \item The horizontal variables \(x,y\) enter only through the data \(B(x,y,z)\) and \(b_s(x,y)\). For each fixed \((x,y)\), the vertical problem is independent.

    \item The formula assumes \(a>0\) and \(\kappa(z)>0\) on \([-H,0]\). Thus one should require
    \[
    a+\beta \exp\!\left[-\frac{z+H}{h}\right] >0
    \]
    throughout the water column.
\end{itemize}
